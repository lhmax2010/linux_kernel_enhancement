   784	sub gbs_export {
   785	    my ($base, $spec, $packaging_dir, $upstream_branch, $upstream_tag, $out_dir) = @_;
   786	    my @args = ();
   787	    my $cmd;
   788	    push @args, "gbs";
   789	    push @args, "--debug" if ($debug);
   790	    push @args, "export";
   791	    push @args, "'$base'";
   792	    push @args, "-o '$out_dir'";
   793	    push @args, "--outdir-directly";
   794	    push @args, "--spec $spec";
   795	    if ($includeall == 1) {
   796	        push @args, "--include-all";
   797	    } else {
   798	        push @args, "--commit=$commit";
   799	    }
   800	    if (! $upstream_branch eq "") {
   801	        push @args, "--upstream-branch='$upstream_branch'";
   802	    }
   803	    if (! $upstream_tag eq "") {
   804	        push @args, "--upstream-tag='$upstream_tag'";
   805	    }
   806	    if ($fallback_to_native == 1) {
   807	        push @args, "--fallback-to-native";
   808	    }
   809	    if (! $squash_patches_until eq "") {
   810	        push @args, "--squash-patches-until=$squash_patches_until";
   811	    }
   812	    if (! $packaging_dir eq "") {
   813	        push @args, "--packaging-dir=$packaging_dir";
   814	    }
   815	    if ($no_patch_export == 1) {
   816	        push @args, "--no-patch-export";
   817	    }
   818	    # print only error messages cause info message appear confused when to use thread
   819	    if ($thread_export == 1){
   820	        push @args, " 2>&1 | grep -v warning | grep -v Creating";
   821	    }
   822	    if ($with_submodules == 1) {
   823	       push @args, "--with-submodules";
   824	    }
   825	
   826	    $cmd = join(" ", @args);
   827	    return my_system($cmd);
   828	}

   840	    my ($cache_key) = @_;

   843	    my $cache = '';
   844	    if (-e $cache_fname) {

   846	            die "read reversion cache($cache_fname) failed: $!";
   847	        $cache = readline($rev);
   848	        close($rev);
   849	        chomp $cache;
   850	    }
   851	    return $cache;
