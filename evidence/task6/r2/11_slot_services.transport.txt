FILE=/etc/update_info_file.sh
#!/bin/sh
# update_info_file.sh : update /etc/info.ini
#

PATH=/bin:/usr/bin:/sbin:/usr/sbin

if [ "$1" == "hal" ]; then
	/usr/bin/system_info_init_db --input=/hal/etc/config/model-config.xml \
		--release=/hal/etc/hal-info.ini \
		--build-info-type=hal \
		--output=/hal/etc/system_info_db
	if [ $? -ne 0 ]; then
		echo "Failed to update system_info_db"
		exit 1
	fi
	echo "Update system info db(/hal/etc/system_info_db) by /hal/etc/config/model-config.xml and /hal/etc/hal-info.ini"
elif [ "$1" == "product" ]; then
	/usr/bin/system_info_init_db --input=/product/etc/config/model-config.xml \
		--release=/product/etc/product-info.ini \
		--build-info-type=product \
		--output=/product/etc/system_info_db
	if [ $? -ne 0 ]; then
		echo "Failed to update system_info_db"
		exit 1
	fi
	echo "Update system info db(/product/etc/system_info_db) by /product/etc/config/model-config.xml and /product/etc/product-info.ini"
else
	/usr/bin/system_info_init_db
	if [ $? -ne 0 ]; then
		echo "Failed to update system_info_db"
		exit 1
	fi

	# Create version info file for update
	if [ -e /usr/share/upgrade/record-version.sh ]; then
		/usr/share/upgrade/record-version.sh
	fi
	echo "Update system info db(/etc/system_info_db) by /etc/config/model-config.xml and /etc/info.ini"
fi
END-RC=0
