FILE drivers/firmware/raspberrypi.c
186: static int rpi_firmware_notify_reboot(struct notifier_block *nb,
187: 				      unsigned long action,
188: 				      void *data)
189: {
190: 	struct rpi_firmware *fw;
191: 	struct platform_device *pdev = g_pdev;
192: 	u32 reboot_flags = 0;
193:
194: 	if (!pdev)
195: 		return 0;
196:
197: 	fw = platform_get_drvdata(pdev);
198: 	if (!fw)
199: 		return 0;
200:
201: 	// The partition id is the first parameter followed by zero or
202: 	// more flags separated by spaces indicating the reason for the reboot.
203: 	//
204: 	// 'tryboot': Sets a one-shot flag which is cleared upon reboot and
205: 	//            causes the tryboot.txt to be loaded instead of config.txt
206: 	//            by the bootloader and the start.elf firmware.
207: 	//
208: 	//            This is intended to allow automatic fallback to a known
209: 	//            good image if an OS/FW upgrade fails.
210: 	//
211: 	// N.B. The firmware mechanism for storing reboot flags may vary
212: 	// on different Raspberry Pi models.
213: 	if (data && strstr(data, " tryboot"))
214: 		reboot_flags |= 0x1;
215:
216: 	// The mailbox might have been called earlier, directly via vcmailbox
217: 	// so only overwrite if reboot flags are passed to the reboot command.
218: 	if (reboot_flags)
219: 		(void)rpi_firmware_property(fw, RPI_FIRMWARE_SET_REBOOT_FLAGS,
220: 				&reboot_flags, sizeof(reboot_flags));
221:
222: 	(void)rpi_firmware_property(fw, RPI_FIRMWARE_NOTIFY_REBOOT, NULL, 0);
223:
224: 	return 0;

101: static void __bcm2835_restart(struct bcm2835_wdt *wdt, u8 partition)
102: {
103: 	u32 val, rsts;
104:
105: 	rsts = (partition & BIT(0)) | ((partition & BIT(1)) << 1) |
106: 	       ((partition & BIT(2)) << 2) | ((partition & BIT(3)) << 3) |
107: 	       ((partition & BIT(4)) << 4) | ((partition & BIT(5)) << 5);
108:
109: 	val = readl_relaxed(wdt->base + PM_RSTS);
110: 	val &= PM_RSTS_PARTITION_CLR;
111: 	val |= PM_PASSWORD | rsts;
112: 	writel_relaxed(val, wdt->base + PM_RSTS);
113:
114: 	/* use a timeout of 10 ticks (~150us) */
115: 	writel_relaxed(10 | PM_PASSWORD, wdt->base + PM_WDOG);
116: 	val = readl_relaxed(wdt->base + PM_RSTC);
117: 	val &= PM_RSTC_WRCFG_CLR;
118: 	val |= PM_PASSWORD | PM_RSTC_WRCFG_FULL_RESET;
119: 	writel_relaxed(val, wdt->base + PM_RSTC);
120:
121: 	/* No sleeping, possibly atomic. */
122: 	mdelay(1);
123: }
