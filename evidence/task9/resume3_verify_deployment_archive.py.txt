from pathlib import Path
import tarfile,json,hashlib
D=Path('/home/linhao/Toolchain/development/linux_kernel_optimization/temp/task9/deploy_lko-base01')
m=json.loads((D/'manifest.json').read_text());expected={x['path']:x for x in m['files']};links={x['path']:x for x in m['symlinks']};seen=set();seen_links=set()
with tarfile.open(D/'deploy_lko-base01.tar.gz','r:gz') as tf:
 for x in tf:
  assert not x.name.startswith('/') and '..' not in Path(x.name).parts
  if x.isfile():
   assert x.name in expected,x.name
   b=tf.extractfile(x).read();assert hashlib.sha256(b).hexdigest()==expected[x.name]['sha256'];assert len(b)==expected[x.name]['bytes'];seen.add(x.name)
  elif x.issym():
   assert x.name in links and x.linkname==links[x.name]['target'];seen_links.add(x.name)
  else:assert x.isdir(),x.name
  assert x.uid==0 and x.gid==0
assert seen==set(expected) and seen_links==set(links)
print('Archive verification PASS: 74 regular files, 1 preserved build symlink; file SHA256/size, symlink target and root uid/gid match; no extraction performed')
