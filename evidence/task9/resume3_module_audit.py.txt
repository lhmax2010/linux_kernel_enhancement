from pathlib import Path
import subprocess,json,hashlib
T=Path('/home/linhao/Toolchain/development/linux_kernel_optimization/temp/task9');V=T/'verify';rel='6.12.80-arm-rpi4-v7l-lko-base01';base=T/'stage/lib/modules'/rel
expected=rel+' SMP mod_unload modversions ARMv7 p2v8'
mods=sorted(base.rglob('*.ko'));assert mods
records=[];errors=[];versions={};compares=[];imports=[]
sym=V/'extract-devel/boot/kernel/devel/kernel-devel-arm-rpi4-lko-base01/Module.symvers'
for line in sym.read_text().splitlines():
 v=line.split();versions[v[1]]=int(v[0],16)
for f in mods:
 cmd=['/usr/bin/qemu-arm','-L','/','/sbin/modinfo','-F','vermagic',str(f)]
 p=subprocess.run(cmd,capture_output=True,text=True)
 item={'file':str(f.relative_to(base)),'command':cmd,'exit':p.returncode,'vermagic':p.stdout.strip(),'stderr':p.stderr,'match':p.returncode==0 and p.stdout.strip()==expected}
 records.append(item)
 if not item['match']:errors.append(item)
for f in sorted((base/'extra').rglob('*.ko')):
 cmd=['/usr/bin/qemu-arm','-L','/','/sbin/modprobe','--show-modversions',str(f)]
 p=subprocess.run(cmd,capture_output=True,text=True)
 (V/('crc_raw_'+f.stem+'.txt')).write_text('COMMAND='+json.dumps(cmd)+'\n'+p.stdout+p.stderr+'\nEXIT='+str(p.returncode)+'\n')
 assert p.returncode==0,(cmd,p.stderr)
 rows=[]
 for line in p.stdout.splitlines():
  fields=line.split();assert len(fields)==2,line
  actual=int(fields[0],16);name=fields[1];expect=versions.get(name)
  row={'module':str(f.relative_to(base)),'symbol':name,'module_crc':f'0x{actual:08x}','kernel_crc':None if expect is None else f'0x{expect:08x}','match':expect==actual}
  rows.append(row);compares.append(row);imports.append((str(f.relative_to(base)),name,row['module_crc']))
 assert rows,f
assert len(list((base/'extra').rglob('*.ko')))==7
summary={'module_count':len(mods),'expected_vermagic':expected,'vermagic_mismatches':errors,'external_modules':7,'crc_comparisons':len(compares),'unique_imported_symbols':len({r['symbol'] for r in compares}),'crc_mismatches':[r for r in compares if not r['match']],'Module.symvers_sha256':hashlib.sha256(sym.read_bytes()).hexdigest()}
(V/'module_vermagic.json').write_text(json.dumps(records,indent=2)+'\n')
(V/'module_crc.json').write_text(json.dumps(compares,indent=2)+'\n')
(V/'module_verification_summary.json').write_text(json.dumps(summary,indent=2)+'\n')
(V/'external_module_imports.tsv').write_text('module\tsymbol\tcrc\n'+''.join('\t'.join(row)+'\n' for row in imports))
print(json.dumps(summary,indent=2));assert not errors and not summary['crc_mismatches']
