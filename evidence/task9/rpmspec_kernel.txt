%build
export AR=llvm-ar RANLIB=llvm-ranlib;
export CC=armv7l-tizen-linux-gnueabi-clang CXX=armv7l-tizen-linux-gnueabi-clang++;
export MAKEFLAGS+=" -k";

export LOCALVERSION=-lko-base01




export LLVM=1


# LSM configuration fragment selection

for target in arm-rpi4 arm-rpi4-rt; do
    echo "-${target}" > localversion-tizen

    # extract uapi headers
    make O=build/${target} headers_install -j40

    # Set config file
    BASE_CONFIG="arch/arm/configs/tizen_bcm2711_defconfig"
    EXTRA_FRAGMENTS=""

    case $target in
	arm-rpi4)
	    EXTRA_FRAGMENTS=""
	;;
	arm-rpi4-rt)
	    EXTRA_FRAGMENTS="kernel/configs/rt.config "
	;;
    esac

    mkdir -p build/${target}
    ./scripts/kconfig/merge_config.sh -m -O build/${target} ${BASE_CONFIG} ${EXTRA_FRAGMENTS}
    make O=build/${target} olddefconfig -j40

    if [ "${target}" = "arm-rpi4" ]; then
        # Build perf
        # WERROR=0 make O=build/${target} -s -C tools/lib/traceevent -j40
        WERROR=0 make O=build/${target} -s -C tools/perf EXTRA_CFLAGS="-fPIE" -j40 NO_LIBTRACEEVENT=1 NO_JEVENTS=1

        # Build bpftool
        WERROR=0 make O=build/${target} -s -C tools/bpf/bpftool -j40
    fi

    # Build Image/Image.gz
    make O=build/${target} -j40

    # Build dtbs
    make O=build/${target} dtbs -j40

    # Build modules
    make O=build/${target} modules -j40

    if [ "${target}" = "arm-rpi4" ]; then
       # Create directory for BTF data
       mkdir -p build/${target}/btf

       # Generate vmlinux.h from vmlinux binary
       ./build/${target}/bpftool btf dump file build/${target}/vmlinux format c > build/${target}/btf/vmlinux.h
    fi

done
