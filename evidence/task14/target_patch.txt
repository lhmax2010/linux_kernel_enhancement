    memory: fix silent failures throughout the reclaim path
    
    Propagate buffered write, flush and close failures while preserving the
    first error, including an EIO fallback when stdio supplies no errno.
    Reject failed integer conversions without modifying the caller output.
    Write a textual compaction trigger and recognize true/false booleans.
    
    Submit vendor force_reclaim commands with one write, accepting positive
    legacy page-count returns without retrying the command. Restrict zram
    algorithm fallback to unavailable algorithms, avoid repeating lz4, and
    retain the kernel-selected algorithm when appropriate. Handle already
    initialized devices and calculate the background swap ratio without
    integer truncation or division by zero.
    
    Only mark swap active after successful activation or an already-active
    device. Iterate safely when removing failed swap modules and when
    releasing notifier links. Initialize failed low-memory vconf reads to
    an unknown state and log all affected read failures consistently.
    Stop boot reclaim when usage does not decrease and use a full-width
    fallback for failed usage reads.
    
    Do not let failure-count persistence errors suppress watchdog recovery.
    Remove incomplete count files, and recover unreadable existing counters
    through the FailOther policy. Preserve recovery action ordering.

+/* This command may return reclaimed pages instead of a byte count on
+ * vendor kernels. Never retry a positive short write as another request.
+ */
+int cgroup_force_reclaim(const char *cgroup_name, uint32_t nr_pages)
+{
+	char path[MAX_PATH_LENGTH];
+	char value[11]; /* UINT32_MAX in decimal, including the terminator. */
+	ssize_t written;
+	int fd, len, ret;
+
+	assert(cgroup_name);
+	len = snprintf(path, sizeof(path), "%s/%s", cgroup_name,
+			MEMCG_FORCE_RECLAIM);
+	if (len < 0 || (size_t)len >= sizeof(path)) {
+		errno = ENAMETOOLONG;
+		ETRACE_ERRNO_MSG("force_reclaim path is too long");
+		return -ENAMETOOLONG;
+	}
+	len = snprintf(value, sizeof(value), "%u", nr_pages);
+
+	fd = open(path, O_WRONLY | O_CLOEXEC);
+	if (fd < 0) {
+		ret = -errno;
+		ETRACE_ERRNO_MSG("Fail to open file %s", path);
+		errno = -ret;
+		return ret;
+	}
+
+	errno = 0;
+	written = write(fd, value, len);
+	ret = written > 0 ? 0 : -(written < 0 ? (errno ?: EIO) : EIO);
+	if (ret < 0) {
+		errno = -ret;
+		ETRACE_ERRNO_MSG("Fail to write force_reclaim request to %s", path);
+	}
+
+	/* A successful write already submitted the command. A close error
+	 * must be reported, but must not turn it into a failed request.
+	 */
+	if (close(fd) < 0) {
+		errno = errno ?: EIO;
+		ETRACE_ERRNO_MSG("Fail to close file %s", path);
+	}
+	errno = ret < 0 ? -ret : 0;
+	return ret;
+}
